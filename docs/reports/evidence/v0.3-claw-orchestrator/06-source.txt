  247     const isResume = !!this.codexThreadId;
  248     const sandbox = this.options.sandboxMode || 'workspace-write';
  249     if (isResume) {
  250       // `codex exec resume` rejects --sandbox and -C (still true as of 0.146.0:
  251       // passing --sandbox errors with "unexpected argument"), so the policy has
  252       // to be restated as a `-c` config override, which resume does accept.
  253       //
  254       // Do NOT rely on the resumed thread inheriting the first turn's sandbox.
  255       // It does not: against 0.146.0, a read-only session whose first turn was
  256       // spawned with `--sandbox read-only` wrote to disk on the very next turn,
  257       // reproducibly, when resume was left to inherit. Restating the mode keeps
  258       // a read-only session read-only for its whole life.
  259       args.push('resume', this.codexThreadId!, '--skip-git-repo-check', '--json');

  261     } else {
  262       args.push('--sandbox', sandbox, '--skip-git-repo-check', '--json');
  263       if (this.options.cwd) args.push('-C', this.options.cwd);
  264     }
  265     // Structured output: Codex 0.132+ accepts `--output-schema <FILE>` on both
  266     // `exec` and `exec resume`, enforcing the model's final response shape.
  267     // The engine-agnostic `jsonSchema` config is inline; Codex needs a path.
  268     const schemaPath = this._ensureSchemaFile();
  269     if (schemaPath) args.push('--output-schema', schemaPath);
  270     if (this.options.model) args.push('--model', this.options.model);
  271     args.push(...this._reasoningEffortArgs());
  272     // `--profile` is rejected by `codex exec resume` (like --sandbox/-C); the
