import {appendFileSync,existsSync,readFileSync} from 'node:fs';
import {definePluginEntry} from 'openclaw/plugin-sdk/plugin-entry';
const logfile='/home/user/v03/openclaw/r3-run/hook-events.jsonl';
function record(value:any){appendFileSync(logfile,JSON.stringify({utc:new Date().toISOString(),pid:process.pid,...value})+'\n');}
export default definePluginEntry({
 id:'r3-hook-probe',name:'R3 Hook Probe',description:'Temporary R3 hook observation and role policy probe.',
 register(api:any){
  record({hook:'register',decisionsEvaluateType:typeof api.runtime?.decisions?.evaluate});
  function context(ctx:any){
   const entry=api.config.agents?.entries?.[ctx.agentId];
   const model=typeof entry?.model==='string'?entry.model:entry?.model?.primary;
   const runtime=entry?.models?.[model]?.agentRuntime?.id??api.config.agents?.defaults?.models?.[model]?.agentRuntime?.id??null;
   return {agentId:ctx.agentId??null,sessionKey:ctx.sessionKey??null,runId:ctx.runId??null,configuredRuntime:runtime,contextRuntime:ctx.runtime??ctx.agentRuntime??null,contextKeys:Object.keys(ctx)};
  }
  api.on('before_tool_call',(event:any,ctx:any)=>{
   const command=event.params?.command??event.params?.cmd??event.params?.script??'';
   const shell=['exec','exec_command','shell','shell_command','local_shell','bash'].includes(String(event.toolName).toLowerCase())&&event.toolKind!=='code_mode_exec'&&ctx.toolKind!=='code_mode_exec';
   const hasDate=JSON.stringify(command).includes('date');
   const block=ctx.agentId==='r3-worker'&&shell&&hasDate;
   record({hook:'before_tool_call',...context(ctx),toolName:event.toolName,toolKind:event.toolKind??null,toolInputKind:event.toolInputKind??null,paramKeys:Object.keys(event.params??{}),shell,hasDate,block});
   if(block)return {block:true,blockReason:'R3 hook probe: date is blocked for r3-worker.'};
  });
  api.on('before_prompt_build',(_event:any,ctx:any)=>{
   const commonFile='/home/user/v03/openclaw/r3-run/common-instructions.txt';
   const common=String(ctx.agentId).startsWith('r3-')&&existsSync(commonFile)?readFileSync(commonFile,'utf8'):'';
   record({hook:'before_prompt_build',...context(ctx),injected:ctx.agentId==='r3-auditor',commonInjected:Boolean(common)});
   const additions=[common,ctx.agentId==='r3-auditor'?'R3-HOOK の語は banana-9':''].filter(Boolean);
   if(additions.length)return {prependContext:additions.join('\n')};
  });
  api.on('before_agent_run',(_event:any,ctx:any)=>{record({hook:'before_agent_run',...context(ctx)});});
  api.on('agent_end',(event:any,ctx:any)=>{record({hook:'agent_end',...context(ctx),success:event.success??null});});
 }
});
