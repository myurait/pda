export function safe(v){
  if(Array.isArray(v))return v.map(safe);
  if(v&&typeof v==='object')return Object.fromEntries(Object.entries(v).map(([k,x])=>[k,/^(deviceTokens?|tokens?|access_token|refresh_token|id_token|password|secret|apiKey|authorization|signature|nonce|publicKey|privateKey(?:Pem)?|recoveryScope)$/i.test(k)?'[REDACTED]':safe(x)]));
  if(typeof v==='string'){
    for(const [id,alias] of Object.entries(aliases))v=v.replaceAll(id,alias);
    for(const x of secrets)if(x)v=v.replaceAll(x,'[REDACTED]');
    return v.replace(/\bsk-[A-Za-z0-9_-]{15,}|\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g,'[REDACTED]').replace(/(Bearer\s+)\S+/gi,'$1[REDACTED]');
  }
  return v;
}
export function identity(alias,persist){
  const path=root+'persistent-identity.json';let d;
  if(persist&&existsSync(path))d=JSON.parse(readFileSync(path,'utf8'));
  else{
    const keys=generateKeyPairSync('ed25519');const pub=keys.publicKey.export({type:'spki',format:'der'}).subarray(-32);
    d={id:createHash('sha256').update(pub).digest('hex'),publicKey:pub.toString('base64url'),privateKey:keys.privateKey.export({type:'pkcs8',format:'pem'}),scopes};
    if(persist)writeFileSync(path,JSON.stringify(d),{mode:0o600,flag:'wx'});
  }
  secrets.add(d.publicKey);secrets.add(d.privateKey);register(d.id,alias);
  d.alias=alias;d.persist=persist;
  if(persist&&existsSync(root+'persistent-token.json'))Object.assign(d,JSON.parse(readFileSync(root+'persistent-token.json','utf8')));
  if(d.token)secrets.add(d.token);
  return d;
}
export class Client{
  constructor(d,label,log){this.identity=d;this.label=label;this.log=log;this.pending=new Map();this.handlers=[];this.closed=false;}
  async connect(){
    const d=this.identity;this.openedAt=new Date().toISOString();this.ws=new WebSocket('ws://127.0.0.1:5114/');
    this.log('socket-opening',{connection:this.label,deviceAlias:d.alias,openedAt:this.openedAt});
    const challenge=await new Promise((resolve,reject)=>{
      const timer=setTimeout(()=>reject(new Error('challenge timeout')),15000);
      this.ws.addEventListener('error',()=>{clearTimeout(timer);reject(new Error('websocket error'));});
      this.ws.addEventListener('close',ev=>{
        this.closed=true;this.closedAt=new Date().toISOString();this.log('socket-closed',{connection:this.label,connId:this.hello?.server?.connId,deviceAlias:d.alias,closedAt:this.closedAt,code:ev.code,reason:ev.reason,wasClean:ev.wasClean});
        for(const p of this.pending.values()){clearTimeout(p.timer);p.reject(new Error(p.method+' socket closed'));}this.pending.clear();
      });
      this.ws.addEventListener('message',ev=>{
        const m=JSON.parse(ev.data);
        if(m.event==='connect.challenge'){clearTimeout(timer);resolve(m.payload);return;}
        if(m.type==='res'){
          const p=this.pending.get(m.id);
          if(p){clearTimeout(p.timer);this.pending.delete(m.id);m.ok?p.resolve(m.payload):p.reject(new Error(p.method+': '+JSON.stringify(m.error)));}
          if(m.payload?.type==='hello-ok')return;
          if(m.payload?.sessions)return;
        }
        this.log('receive',{connection:this.label,frame:m});for(const h of this.handlers)h(m);
      });
    });
    const grant=d.scopes??scopes;const token=d.token??'';
    const payload=['v3',d.id,'cli','cli','operator',grant.join(','),String(challenge.ts),token,challenge.nonce,'linux',''].join('|');
    const params={minProtocol:4,maxProtocol:4,client:{id:'cli',version:'v03-authority-2026-10-11',platform:'linux',mode:'cli',displayName:'PDA authority check 2026-10-11'},role:'operator',scopes:grant,caps:['tool-events'],device:{id:d.id,publicKey:d.publicKey,signature:sign(null,Buffer.from(payload),d.privateKey).toString('base64url'),signedAt:challenge.ts,nonce:challenge.nonce},...(token?{auth:{token}}:{})};
    const h=await this.request('connect',params);this.hello=h;this.connectedAt=new Date().toISOString();
    this.log('connected',{connection:this.label,deviceAlias:d.alias,protocol:h.protocol,server:h.server,scopes:h.auth?.scopes,usedStoredToken:Boolean(token),connectedAt:this.connectedAt,client:params.client});
    if(h.auth?.deviceToken){
      secrets.add(h.auth.deviceToken);
      const newToken=h.auth.deviceToken!==d.token;d.token=h.auth.deviceToken;
      if(newToken)d.scopes=h.auth.scopes;
      if(d.persist)writeFileSync(root+'persistent-token.json',JSON.stringify({token:d.token,scopes:d.scopes}),{mode:0o600});
    }
    return h;
  }
  request(method,params={}){
    const id=randomUUID();
    if(method==='connect')this.log('send',{connection:this.label,frame:{type:'req',id,method,params:{client:params.client,role:params.role,scopes:params.scopes,deviceAlias:this.identity.alias,usedStoredToken:Boolean(params.auth?.token)}}});
    else if(method!=='sessions.list')this.log('send',{connection:this.label,frame:{type:'req',id,method,params}});
    return new Promise((resolve,reject)=>{
      if(this.ws.readyState!==WebSocket.OPEN){reject(new Error(method+' socket not open'));return;}
      const timer=setTimeout(()=>{this.pending.delete(id);reject(new Error(method+' timeout'));},30000);
      this.pending.set(id,{resolve,reject,timer,method});this.ws.send(JSON.stringify({type:'req',id,method,params}));
    });
  }
  async close(reason){
    if(this.closed)return;
    this.log('socket-close-requested',{connection:this.label,connId:this.hello?.server?.connId,reason});
    await new Promise(resolve=>{const t=setTimeout(resolve,3000);this.ws.addEventListener('close',()=>{clearTimeout(t);resolve();},{once:true});this.ws.close(1000,reason);});
  }
}

def scrub_text(s):
    for key, alias in registry().items(): s = s.replace(key, alias)
    s = re.sub(r'\bsk-[A-Za-z0-9_-]{15,}|\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+', '[REDACTED]', s)

TARGETS=['Gateway client authority closed','Gateway caller authority is no longer active','Subagent terminal signal owner changed before commit']
def iso(ms): return datetime.datetime.fromtimestamp(ms/1000,datetime.timezone.utc).isoformat() if isinstance(ms,(int,float)) else ms
def content(m):
    c=m.get('content',[])
    return c if isinstance(c,str) else '\n'.join(b.get('text','') for b in c if b.get('type')=='text')
def records(name): return [json.loads(x) for x in (EV/name).read_text().splitlines()]
def analyze(label):
    s=json.loads((EV/(label+'-summary.json')).read_text()); events=records(label+'.jsonl')
    answers=[]; tool_errors=[]; calls={}; hist_targets=[]
    for h in s['histories']:
        for m in h['history'].get('messages',[]):
            text=content(m)
            if m.get('role')=='assistant' and text: answers.append({'key':h['key'],'timestamp':iso(m.get('timestamp')),'text':text})
            blocks=m.get('content',[])
            if isinstance(blocks,list):
                for b in blocks:
                    if b.get('type') in ['toolcall','tool_use']: calls[b.get('id')]={'name':b.get('name'),'arguments':b.get('arguments',b.get('input'))}
                for b in blocks:
                    if b.get('type') in ['tool_result','toolResult'] and (b.get('is_error') or b.get('isError') or any(t in json.dumps(b) for t in TARGETS)):
                        tool_errors.append({'source':'history','key':h['key'],'timestamp':iso(m.get('timestamp')),'toolCallId':b.get('tool_use_id',b.get('toolCallId')),**calls.get(b.get('tool_use_id',b.get('toolCallId')),{}),'result':b})
            if any(t in json.dumps(m,ensure_ascii=False) for t in TARGETS): hist_targets.append({'key':h['key'],'timestamp':iso(m.get('timestamp')),'message':m})
    event_errors=[]
    for i,r in enumerate(events,1):
        f=r['data'].get('frame',{}); p=f.get('payload',{}); d=p.get('data',{})
        if f.get('event')=='agent' and p.get('stream')=='tool' and d.get('phase')=='result' and (d.get('isError') or any(t in json.dumps(d) for t in TARGETS)):
            event_errors.append({'sourceFile':label+'.jsonl','sourceLine':i,'utc':r['utc'],'eventTs':p.get('ts'),'key':p.get('sessionKey'),'runId':p.get('runId'),**d})
    source_rows={}
    for name in [label+'.jsonl',label+'-journal.txt',label+'-gateway-log.jsonl']:
        if not (EV/name).exists():continue
        source_rows[name]={t:[{'line':i,'text':line} for i,line in enumerate((EV/name).read_text().splitlines(),1) if t in line] for t in TARGETS}
    child_answers=[a for a in answers if a['key']!=s['key']]; parents=[a for a in answers if a['key']==s['key']]
    numeric=[a for a in parents if child_answers and re.search(r'(?<!\d)15(?!\d)',a['text'])]
    full=[]
    if child_answers:
        raw=child_answers[-1]['text']
        for a in parents:
            normalized=re.sub(r'^\s*>\s?','',a['text'],flags=re.M)
            if raw in normalized: full.append(a)
    target_tool_errors=[e for e in tool_errors if any(t in json.dumps(e,ensure_ascii=False) for t in TARGETS)]
    rpcs=sorted(set(re.findall(r'Gateway client authority closed before dispatching ([\w.]+)',json.dumps(source_rows,ensure_ascii=False))))
    result={'label':label,'key':s['key'],'startedAt':s['startedAt'],'sentAt':s.get('sentAt'),'endedAt':s['endedAt'],'connections':s['connections'],'disconnectedAt':s.get('disconnectedAt'),'reconnectedAt':s.get('reconnectedAt'),'disconnectedMs':s.get('disconnectedMs'),'children':s['children'],'answers':answers,'numericAnswerQuoted':bool(numeric),'entireChildMessageQuoted':bool(full),'firstNumericQuoteAt':numeric[0]['timestamp'] if numeric else None,'firstWholeQuoteAt':full[0]['timestamp'] if full else None,'secondsToNumericQuote':(datetime.datetime.fromisoformat(numeric[0]['timestamp'])-datetime.datetime.fromisoformat(s['sentAt'].replace('Z','+00:00'))).total_seconds() if numeric else None,'toolErrorsInHistory':tool_errors,'toolErrorsInEvents':event_errors,'historyMessagesWithTargetErrors':hist_targets,'targetErrorOccurrencesBySource':source_rows,'failedRPCs':rpcs,'stopReason':s.get('stopReason'),'collectorFailures':s['failures']}
    write(label+'-analysis.json',result)
    compact={k:result[k] for k in ['label','numericAnswerQuoted','entireChildMessageQuoted','firstNumericQuoteAt','secondsToNumericQuote','failedRPCs','stopReason']}
    compact['targetCountsInJournal']={t:len(x) for t,x in source_rows.get(label+'-journal.txt',{}).items()}
    compact['historyToolErrorCount']=len(tool_errors)
    return compact
if __name__=='__main__':
    labels=sys.argv[1:] or [p.name.removesuffix('-summary.json') for p in sorted(EV.glob('*-summary.json'))]
    results=[analyze(label) for label in labels]
    print(json.dumps(results,ensure_ascii=False,indent=2))

from manage import *

checks=[]
def check(name, value):
    checks.append({'name':name,'pass':bool(value)})
    if not value: raise AssertionError(name)
def read(name): return json.loads((EV/name).read_text())
def rows(name): return [json.loads(x) for x in (EV/name).read_text().splitlines()]
def timestamp(s): return datetime.datetime.fromisoformat(s.replace('Z','+00:00'))

try:
    labels=['A1','A2','A2-retry','A3','A3-retry','B','C']
    summaries={label:read(label+'-summary.json') for label in labels}
    prompt='r3-worker に sessions_spawn で「7 と 8 を足した答えだけを 1 行で返して」と頼み、子の完了の通知を待ってから、子の答えをそのまま引用して私に報告して。'
    check('All attempts use exactly the requested prompt',all(s['prompt']==prompt for s in summaries.values()))
    check('Every attempt has its own new r3-leader session',len({s['key'] for s in summaries.values()})==len(labels) and all(s['key'].startswith('agent:r3-leader:authority-') for s in summaries.values()))
    check('Only A2 and A3 retried, once each',sorted(p.name for p in EV.glob('*-summary.json'))==sorted(label+'-summary.json' for label in labels))
    for label,s in summaries.items():
        r=rows(label+'.jsonl')
        sends=[x for x in r if x['type']=='send' and x['data'].get('frame',{}).get('method')=='chat.send']
        check(label+' has exactly one chat.send',len(sends)==1)
        check(label+' collector completed without transport/RPC failure',not s['failures'])
        check(label+' parent history obtained',any(h['key']==s['key'] for h in s['histories']))
        check(label+' journal and file log collected',read(label+'-log-window.json')['journalExitCode']==0 and (EV/(label+'-journal.txt')).exists() and (EV/(label+'-gateway-log.jsonl')).exists())
        check(label+' all own sessions idle at end',s['polls'] and all(not x.get('hasActiveRun') for x in s['polls'][-1]['statuses']))
        check(label+' initial connection uses saved identity',s['connections'][0]['deviceAlias']=='authority-persistent')
    for label in ['A1','A2','A2-retry','A3','A3-retry']:
        s=summaries[label]
        check(label+' one uninterrupted socket',len(s['connections'])==1 and len([x for x in rows(label+'.jsonl') if x['type']=='socket-closed'])==1)
        finals=[x for x in s['terminalEvents'] if x['key']==s['key'] and x['state']=='final']
        check(label+' connected until final reply',bool(finals) and timestamp(s['connections'][0]['closedAt'])>=max(timestamp(x['utc']) for x in finals))
        check(label+' final reply arrived within five minutes',max(timestamp(x['utc']) for x in finals)-timestamp(s['sentAt'])<=datetime.timedelta(minutes=5))
    for label in ['A1','A2-retry','A3-retry']:
        s=summaries[label]; a=read(label+'-analysis.json')
        check(label+' actual worker completion and numeric quote recorded',len(s['children'])==1 and a['numericAnswerQuoted'])
    for label in ['A2','A3']:
        a=read(label+'-analysis.json')
        check(label+' initial model-unavailable failure retained',not summaries[label]['children'] and 'The selected account or native runtime is unavailable' in json.dumps(a['toolErrorsInHistory']))
    for label in ['B','C']:
        s=summaries[label];c=s['connections'];r=rows(label+'.jsonl')
        check(label+' exactly one disconnect and one reconnect',len(c)==2 and c[0]['connId']!=c[1]['connId'])
        check(label+' disconnect triggered by observed sessions_spawn',s['disconnectTriggeredBy']['name'].endswith('sessions_spawn') and timestamp(c[0]['closedAt'])>=timestamp(s['disconnectTriggeredBy']['utc']))
        check(label+' waited at least 120 seconds before reconnect',s['disconnectedMs']>=120000)
        check(label+' reads history after reconnect',all(timestamp(h['retrievedAt'])>=timestamp(s['reconnectedAt']) for h in s['histories']))
        check(label+' one initial task; no resend on reconnect',sum(x['type']=='send' and x['data'].get('frame',{}).get('method')=='chat.send' for x in r)==1)
    check('B reconnect uses same saved identity and token',summaries['B']['connections'][1]['deviceAlias']=='authority-persistent' and summaries['B']['connections'][1]['usedStoredToken'])
    check('C reconnect uses new in-memory identity without saved token',summaries['C']['connections'][1]['deviceAlias']=='authority-ephemeral' and summaries['C']['connections'][1]['usedStoredToken'] is False)
    check('Identity files are outside repository with mode 0600',all((RUN/n).stat().st_mode&0o777==0o600 for n in ['persistent-identity.json','persistent-token.json']))
    check('No ephemeral keypair file',not any('ephemeral' in p.name for p in RUN.iterdir()))
    check('Exactly one saved identity reused and one new C identity',list(registry().values()).count('authority-persistent')==1 and list(registry().values()).count('authority-ephemeral')==1)
    check('Configuration, agent instructions, plugin, TypeSafe unchanged',inventory()==read('baseline.json')['unchangedFiles'])
    cleanup=read('cleanup.json')
    check('Only run-created devices targeted for removal',all(r['deviceAlias'].startswith('authority-') for r in cleanup['removals']))
    check('All run devices removed; existing devices retained',cleanup['allOwnAbsent'] and cleanup['allPreexistingPreserved'])
    before={d['deviceId'] for d in read('devices-before.json')['result']['paired']};after={d['deviceId'] for d in read('devices-after-cleanup.json')['result']['paired']}
    check('Final paired device list equals initial list',before==after)
    check('OpenClaw active/running, preview and readyz HTTP 200','ActiveState=active' in cleanup['service']['stdout'] and 'SubState=running' in cleanup['service']['stdout'] and all(r['stdout']=='200' for r in cleanup['http']))
    check('No other v03 service running',all('v03-openclaw.service' in line for line in cleanup['allProductUnits']['stdout'].splitlines() if re.search(r'\bactive\s+running\b',line)))
    for p in sorted(EV.glob('restart*.json')):
        r=json.loads(p.read_text())
        check(p.name+' records reason, time, successful restart',bool(r['reason']) and bool(r['requestedAt']) and r['restart']['exitCode']==0 and 'ActiveState=active' in r['after']['stdout'])
    old=subprocess.check_output(['git','show','0dd6db0d976f4a58d8f21cd45d3cab1a39298134:docs/reports/v0.3-openclaw-codex.md'],cwd=REPO)
    report=(REPO/'docs/reports/v0.3-openclaw-codex.md').read_bytes()
    check('Prior report sections 1-10 byte-for-byte unchanged',report.startswith(old))
    tail=report[len(old):].decode()
    check('Exactly the requested section 11 appended',tail.count('## 11. 子の結果を受け取る段の誤りの確認 (2026-10-11)')==1)
    headings=re.findall(r'^### 11\.[1-6] (.+)$',tail,re.M)
    check('Requested report order',headings==['結論','版','試し A〜C の結果','出る条件','後片付け','分からなかったこと'])
    check('Report discloses model failures and recovery restarts',all(t in tail for t in ['A2','A3','再起動','available=false']))
    check('Report distinguishes answer quote from full child message',all(t in tail for t in ['[ROLE:worker]','全文','15']))
    check('Evidence links resolve; verification.json is this run output',all((REPO/'docs/reports'/x).exists() or REPO/'docs/reports'/x==EV/'verification.json' for x in re.findall(r'\]\((evidence/v0\.3-openclaw/2026-10-11-authority/[^)]+)\)',tail)))
    check('Collector scripts remain outside repository',not any(p.suffix in ['.py','.mjs','.js'] for p in EV.iterdir()))
    alltext='\n'.join(p.read_text(errors='replace') for p in EV.iterdir() if p.is_file())+'\n'+tail
    secrets=list(registry())
    ident=json.loads((RUN/'persistent-identity.json').read_text());token=json.loads((RUN/'persistent-token.json').read_text())
    secrets += [ident['publicKey'],ident['privateKey'],''.join(ident['privateKey'].splitlines()[1:-1]),token['token']]
    check('Actual device IDs, keys and saved token absent from evidence/report',all(v not in alltext for v in secrets if v))
    check('No credential-looking values',not re.search(r'\bsk-[A-Za-z0-9_-]{15,}|\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+|-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----',alltext))
    check('Correct branch',subprocess.check_output(['git','branch','--show-current'],cwd=REPO,text=True).strip()=='v0.3/authority-openclaw')
    check('Started from requested main base',subprocess.run(['git','merge-base','--is-ancestor','0dd6db0d976f4a58d8f21cd45d3cab1a39298134','HEAD'],cwd=REPO).returncode==0)
    check('git diff --check',subprocess.run(['git','diff','--check'],cwd=REPO).returncode==0)
    for p in EV.iterdir():
        if p.suffix=='.json':json.loads(p.read_text())
        elif p.suffix=='.jsonl':rows(p.name)
    check('All JSON/JSONL evidence parses',True)
except Exception as e:
    checks.append({'name':'ERROR '+str(e),'pass':False})
finally:
    write('verification.json',{'utc':now(),'meaning':'Completion of requested experiment and faithful reporting, including product errors and explicitly recorded recoveries; not a product correctness claim.','checks':checks,'passed':all(c['pass'] for c in checks)})
    print('\n'.join(('PASS ' if c['pass'] else 'FAIL ')+c['name'] for c in checks))
    if not all(c['pass'] for c in checks):raise SystemExit(1)
