const aliases=JSON.parse(readFileSync(root+'device-aliases.json','utf8'));

function register(id,alias){aliases[id]=alias;writeFileSync(root+'device-aliases.json',JSON.stringify(aliases,null,2),{mode:0o600});}
export function safe(v){
  if(Array.isArray(v))return v.map(safe);
  if(v&&typeof v==='object')return Object.fromEntries(Object.entries(v).map(([k,x])=>[k,/^(deviceTokens?|tokens?|access_token|refresh_token|id_token|password|secret|apiKey|authorization|signature|nonce|publicKey|privateKey(?:Pem)?|recoveryScope)$/i.test(k)?'[REDACTED]':safe(x)]));
  if(typeof v==='string'){
    for(const [id,alias] of Object.entries(aliases))v=v.replaceAll(id,alias);
    for(const x of secrets)if(x)v=v.replaceAll(x,'[REDACTED]');
    return v.replace(/\bsk-[A-Za-z0-9_-]{15,}|\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g,'[REDACTED]').replace(/(Bearer\s+)\S+/gi,'$1[REDACTED]');
  }
  return v;
}
export function identity(alias,persist){
  const path=root+'persistent-identity.json';let d;
  if(persist&&existsSync(path))d=JSON.parse(readFileSync(path,'utf8'));
  else{
    const keys=generateKeyPairSync('ed25519');const pub=keys.publicKey.export({type:'spki',format:'der'}).subarray(-32);
    d={id:createHash('sha256').update(pub).digest('hex'),publicKey:pub.toString('base64url'),privateKey:keys.privateKey.export({type:'pkcs8',format:'pem'}),scopes};
    if(persist)writeFileSync(path,JSON.stringify(d),{mode:0o600,flag:'wx'});
  }
  secrets.add(d.publicKey);secrets.add(d.privateKey);register(d.id,alias);
  d.alias=alias;d.persist=persist;
  if(persist&&existsSync(root+'persistent-token.json'))Object.assign(d,JSON.parse(readFileSync(root+'persistent-token.json','utf8')));
  if(d.token)secrets.add(d.token);
  return d;
}
export class Client{
  constructor(d,label,log){this.identity=d;this.label=label;this.log=log;this.pending=new Map();this.handlers=[];this.closed=false;}
  async connect(){
    const d=this.identity;this.openedAt=new Date().toISOString();this.ws=new WebSocket('ws://127.0.0.1:5114/');
    this.log('socket-opening',{connection:this.label,deviceAlias:d.alias,openedAt:this.openedAt});
    const challenge=await new Promise((resolve,reject)=>{
      const timer=setTimeout(()=>reject(new Error('challenge timeout')),15000);
      this.ws.addEventListener('error',()=>{clearTimeout(timer);reject(new Error('websocket error'));});
      this.ws.addEventListener('close',ev=>{
        this.closed=true;this.closedAt=new Date().toISOString();this.log('socket-closed',{connection:this.label,connId:this.hello?.server?.connId,deviceAlias:d.alias,closedAt:this.closedAt,code:ev.code,reason:ev.reason,wasClean:ev.wasClean});
        for(const p of this.pending.values()){clearTimeout(p.timer);p.reject(new Error(p.method+' socket closed'));}this.pending.clear();
      });
      this.ws.addEventListener('message',ev=>{
        const m=JSON.parse(ev.data);
        if(m.event==='connect.challenge'){clearTimeout(timer);resolve(m.payload);return;}
        if(m.type==='res'){
          const p=this.pending.get(m.id);
          if(p){clearTimeout(p.timer);this.pending.delete(m.id);m.ok?p.resolve(m.payload):p.reject(new Error(p.method+': '+JSON.stringify(m.error)));}
          if(m.payload?.type==='hello-ok')return;
          if(m.payload?.sessions)return;
        }
        this.log('receive',{connection:this.label,frame:m});for(const h of this.handlers)h(m);
      });
    });
    const grant=d.scopes??scopes;const token=d.token??'';
    const payload=['v3',d.id,'cli','cli','operator',grant.join(','),String(challenge.ts),token,challenge.nonce,'linux',''].join('|');
    const params={minProtocol:4,maxProtocol:4,client:{id:'cli',version:'v03-roles-2026-10-11',platform:'linux',mode:'cli',displayName:'PDA roles check 2026-10-11'},role:'operator',scopes:grant,caps:['tool-events'],device:{id:d.id,publicKey:d.publicKey,signature:sign(null,Buffer.from(payload),d.privateKey).toString('base64url'),signedAt:challenge.ts,nonce:challenge.nonce},...(token?{auth:{token}}:{})};
    const h=await this.request('connect',params);this.hello=h;this.connectedAt=new Date().toISOString();
    this.log('connected',{connection:this.label,deviceAlias:d.alias,protocol:h.protocol,server:h.server,scopes:h.auth?.scopes,usedStoredToken:Boolean(token),connectedAt:this.connectedAt,client:params.client});
    if(h.auth?.deviceToken){
      secrets.add(h.auth.deviceToken);
      const newToken=h.auth.deviceToken!==d.token;d.token=h.auth.deviceToken;
      if(newToken)d.scopes=h.auth.scopes;
      if(d.persist)writeFileSync(root+'persistent-token.json',JSON.stringify({token:d.token,scopes:d.scopes}),{mode:0o600});
    }
    return h;
  }
  request(method,params={}){
    const id=randomUUID();
    if(method==='connect')this.log('send',{connection:this.label,frame:{type:'req',id,method,params:{client:params.client,role:params.role,scopes:params.scopes,deviceAlias:this.identity.alias,usedStoredToken:Boolean(params.auth?.token)}}});
    else if(method!=='sessions.list')this.log('send',{connection:this.label,frame:{type:'req',id,method,params}});
    return new Promise((resolve,reject)=>{
      if(this.ws.readyState!==WebSocket.OPEN){reject(new Error(method+' socket not open'));return;}
      const timer=setTimeout(()=>{this.pending.delete(id);reject(new Error(method+' timeout'));},30000);
      this.pending.set(id,{resolve,reject,timer,method});this.ws.send(JSON.stringify({type:'req',id,method,params}));
    });
  }
  async close(reason){
    if(this.closed)return;
    this.log('socket-close-requested',{connection:this.label,connId:this.hello?.server?.connId,reason});
    await new Promise(resolve=>{const t=setTimeout(resolve,3000);this.ws.addEventListener('close',()=>{clearTimeout(t);resolve();},{once:true});this.ws.close(1000,reason);});
  }
}
